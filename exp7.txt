import requests
# Reusable function to connect Python with Ollama
def ask_ollama(prompt):
 url = "http://localhost:11434/api/generate"
 data = {
 "model": "llama3.2",
 "prompt": prompt,
 "stream": False
 }
 response = requests.post(url, json=data)
 response.raise_for_status()
 return response.json()["response"]
# Reusable function to display the AI response
def display_result(result):
 print("\n----- AI OUTPUT -----\n")
 print(result)
# Reusable function to save the AI response in a text file
def save_result(result, filename):
 with open(filename, "w", encoding="utf-8") as file:
 file.write(result)
 print("\nResult saved in", filename)
try:
 with open("resume.txt", "r", encoding="utf-8") as file:
 resume = file.read()
except FileNotFoundError:
 print("Error: Create resume.txt in the same folder as this program.")
 raise SystemExit
prompt = f"""Analyze this resume.
Identify the candidate's skills and qualifications.
Generate a brief resume summary.
Suggest areas for improvement.
Resume:
{resume}"""
result = ask_ollama(prompt)
display_result(result)
